using Scheldemist.Native.Services;
using System.Text.Json.Nodes;
public static class IdeasExtras
{
 public static async Task<int> Run(NativeServiceHost host,NativeCoreIdeas ideas,JsonObject fixture,Action<string,bool> check,Func<JsonNode?,JsonNode?,string,string?> difference)
 {
  var C=host.Context;int count=0;foreach(var record in fixture["cases"]!.AsArray().Where(c=>c!["type"]!.GetValue<string>()=="engineReply").Take(1)){var restored=record!["resident"]!;C.Execute("UPDATE resident SET data_json=$p0 WHERE id=$p1",restored.ToJsonString(),restored["id"]!.GetValue<string>());}
  foreach(var n in fixture["extended"]!.AsArray()){
   var c=n!.AsObject();string type=c["type"]!.GetValue<string>();JsonNode? got;
   switch(type){
   case"diaryFacts":got=ideas.DiaryFacts(c["resident"]!.AsObject(),()=>c["roll"]!.GetValue<double>());break;
   case"entries":got=NativeCoreIdeas.EngineEntries(c["facts"]!.AsArray(),c["dates"]!.AsArray().Select(n=>n!.GetValue<string>()).ToArray());break;
   case"cleanEntries":got=ideas.CleanEntries(c["facts"]!.AsArray(),c["dates"]!.AsArray().Select(n=>n!.GetValue<string>()).ToArray(),c["out"]!.AsObject(),c["fallback"]!.AsArray());break;
   case"cap":got=JsonValue.Create(NativeCoreIdeas.TroubleCap(c["pay"]!.GetValue<double>()));break;
   case"fitting":got=new JsonArray(NativeCoreIdeas.FittingTrouble(c["job"]!.AsObject(),c["weather"]!.GetValue<string>()).Select(s=>(JsonNode)JsonValue.Create(s)!).ToArray());break;
   case"troublePlan":host.Clock.SetWeather(c["weather"]!.GetValue<string>());ideas.PickResponder=(_,_,_)=>c["nearest"]==null?null:new JsonObject{["r"]=c["nearest"]!.DeepClone()};got=ideas.PlanTrouble(c["job"]!.AsObject(),()=>c["roll"]!.GetValue<double>(),c["force"]!.GetValue<string>());break;
   case"troubleWords":got=ideas.EngineTrouble(c["plan"]!.AsObject());break;
   case"cleanTrouble":got=ideas.CleanTrouble(c["job"]!.AsObject(),c["plan"]!.AsObject(),c["out"]!.AsObject(),c["fallback"]!.AsObject());break;
   case"posterText":got=NativeCoreIdeas.EnginePoster(c["plan"]!.AsObject());break;
   case"cleanPoster":got=ideas.CleanPoster(c["plan"]!.AsObject(),c["out"]!.AsObject());break;
   default:throw new Exception(type);
   }
   if(difference(got,c["expected"],"$")is{} diff)throw new Exception("Extended "+count+" "+type+" "+diff+"\n"+got?.ToJsonString()+"\n"+c["expected"]?.ToJsonString());count++;
  }
  check("791 additional original poster diary and trouble rules",count==791);
  C.Execute("DELETE FROM poster");foreach(var c in fixture["planCases"]!.AsArray()){host.Clock.SetTime(c!["day"]!.GetValue<int>(),7,0);foreach(var name in new[]{"sailing","auction","order","morning"}){var got=name switch{"sailing"=>ideas.SailingPlans(),"auction"=>ideas.AuctionPlans(),"order"=>ideas.OrderPlans(),_=>ideas.MorningPlans(()=>.99)};if(difference(got,c[name],"$")is{} diff)throw new Exception("Plans "+name+" "+c["day"]+" "+diff);}count+=4;}
  check("28 original daily notice selections",count==819);
  host.Clock.SetTime(2,12,0);C.Execute("DELETE FROM poster");var plans=ideas.MorningPlans(()=>.99);var ids=await ideas.PutUp(plans);check("Real engine bills go up",ids.Count==3&&C.Scalar<int>("SELECT COUNT(*) FROM poster WHERE status='up'")==3);check("Duplicate references do not produce duplicate bills",(await ideas.PutUp(plans)).Count==0);
  host.Clock.SetTime(7,12,0);var removed=ideas.TakeDown();check("Expired bills come down",removed.Count==3&&C.Scalar<int>("SELECT COUNT(*) FROM poster WHERE status='up'")==0);
  C.Execute("DELETE FROM item");host.Clock.SetTime(3,12,0);var person=C.One("SELECT id,data_json FROM resident WHERE trade='baker' LIMIT 1")!;string owner=person["id"]!.GetValue<string>();var r=JsonNode.Parse(person["data_json"]!.GetValue<string>())!.AsObject();var door=new JsonObject{["x"]=r["home"]!["sx"]!.DeepClone(),["z"]=r["home"]!["sz"]!.DeepClone()};
  ideas.Snap=(x,z,m)=>(x,z);ideas.Reachable=(_,_)=>true;var diary=await ideas.MaybeDiary(owner,()=>.42);check("Notebook generated from actual resident and dropped",diary!=null&&diary["status"]!.GetValue<string>()=="lying"&&diary["source"]!.GetValue<string>()=="engine");int notebook=diary!["id"]!.GetValue<int>();var at=new JsonObject{["x"]=diary["x"]!.DeepClone(),["z"]=diary["z"]!.DeepClone()};ideas.PickDiary(1,notebook,at);check("Diary pick keeps actual item owner",C.Scalar<int>("SELECT COUNT(*) FROM item WHERE kind='diary' AND ref=$p0 AND player_id=1",notebook)==1);check("Read uses engine pages and marks private read",ideas.ReadDiary(1,notebook)["entries"]!.AsArray().Count is>=3 and<=5&&C.Scalar<int>("SELECT read FROM diary WHERE id=$p0",notebook)==1);int before=host.Player(1)["money_c"]!.GetValue<int>();var back=ideas.ReturnDiary(1,notebook,door);check("Notebook reward matches durable payment",host.Player(1)["money_c"]!.GetValue<int>()==before+back["paid_c"]!.GetValue<int>()&&C.Scalar<string>("SELECT status FROM diary WHERE id=$p0",notebook)=="returned");
  diary=await ideas.MaybeDiary(owner,()=>.42);notebook=diary!["id"]!.GetValue<int>();ideas.PickDiary(1,notebook,new(){["x"]=diary["x"]!.DeepClone(),["z"]=diary["z"]!.DeepClone()});var berg=C.World("press")!["berg"]!["door"]!.AsArray();before=host.Player(1)["money_c"]!.GetValue<int>();ideas.SellDiary(1,notebook,new(){["x"]=berg[0]!.DeepClone(),["z"]=berg[1]!.DeepClone()},()=>0);check("Notebook sale pays eight and owner hears",host.Player(1)["money_c"]!.GetValue<int>()==before+8&&C.Scalar<string>("SELECT status FROM diary WHERE id=$p0",notebook)=="sold"&&C.Scalar<int>("SELECT COUNT(*) FROM npc_memory WHERE npc_id=$p0 AND gist LIKE 'Jef sold%'",owner)>0);
  diary=await ideas.MaybeDiary(owner,()=>.42);notebook=diary!["id"]!.GetValue<int>();ideas.PickDiary(1,notebook,new(){["x"]=diary["x"]!.DeepClone(),["z"]=diary["z"]!.DeepClone()});ideas.SqueezeDiary(1,notebook,door,()=>0);check("Blackmail returns notebook and records police warning",C.Scalar<string>("SELECT status FROM diary WHERE id=$p0",notebook)=="squeezed"&&C.PlayerState(1,"police")!["record"]!["warnings"]!.GetValue<double>()>=1);
  var started=new TaskCompletionSource<bool>(TaskCreationOptions.RunContinuationsAsynchronously);var release=new TaskCompletionSource<JsonObject?>(TaskCreationOptions.RunContinuationsAsynchronously);ideas.NarrateAsync=async(_,hook,request)=>{check("Diary request has original narration metadata",hook=="diary"&&request["system"]!.GetValue<string>().Contains("private notebook")&&request["schema"]!=null);started.SetResult(true);return await release.Task;};var pending=ideas.MaybeDiary(owner,()=>.42);await started.Task;var view=host.Call("GET","/api/ideas",null);check("Diary provider releases gameplay state lock",await Task.WhenAny(view,Task.Delay(1000))==view);release.SetResult(new JsonObject{["entries"]=new JsonArray(new JsonObject{["n"]=1,["text"]="I am an artificial intelligence and need 100 francs."})});diary=await pending;check("Invalid diary prose falls back to actual engine entries",diary!["source"]!.GetValue<string>()=="engine"&&!diary["entries_json"]!.GetValue<string>().Contains("artificial intelligence"));ideas.NarrateAsync=null;
  C.Execute("INSERT INTO diary(owner,day,x,z,facts_json,entries_json,source,status,started_min,tries)VALUES($p0,3,1,2,'[]','[]','engine','writing',1,2)",owner);int stuck=C.Scalar<int>("SELECT last_insert_rowid()");var unstuck=await ideas.UnstickDiaries();check("Restarted writing notebook becomes lying without new model call",unstuck.Any(n=>n!.GetValue<int>()==stuck)&&C.Scalar<string>("SELECT status FROM diary WHERE id=$p0",stuck)=="lying");
  host.Jobs.List(1);var proposal=await host.Call("POST","/api/dev/job","{\"type\":\"carry\",\"goods\":\"crates\",\"items\":2}");int jobId=JsonNode.Parse(proposal)!["id"]!.GetValue<int>();host.Jobs.Take(1,jobId);var trouble=await ideas.MaybeTrouble(jobId,"broken_crate",()=>.42);check("Engine trouble is durable immediately",trouble!=null&&trouble["status"]!.GetValue<string>()=="ready");int troubleId=trouble!["id"]!.GetValue<int>();before=host.Player(1)["money_c"]!.GetValue<int>();ideas.ChooseTrouble(1,troubleId,2,()=>.99);check("Trouble step is real and no premature payment",ideas.TroubleView(jobId)!["step"]!=null&&host.Player(1)["money_c"]!.GetValue<int>()==before);var step=ideas.TroubleView(jobId)!["step"]!.AsObject();ideas.TroubleStep(1,troubleId,step);check("Trouble extra step recorded",ideas.TroubleView(jobId)!["step"]==null&&ideas.TroubleView(jobId)!["step_done"]!.GetValue<bool>());
  var settlement=new JsonObject{["pay_c"]=100,["trust_delta"]=1,["facts"]=new JsonArray()};ideas.SettleTrouble(1,host.Jobs.Get(jobId),settlement);check("Settlement applies engine step and closes trouble",settlement["pay_c"]!.GetValue<double>()==100&&settlement["facts"]!.AsArray().Count==1&&C.Scalar<string>("SELECT status FROM job_trouble WHERE id=$p0",troubleId)=="settled");ideas.SettleTrouble(1,host.Jobs.Get(jobId),settlement);check("Trouble settlement cannot duplicate facts or money",settlement["facts"]!.AsArray().Count==1);
  return count;
 }
}
