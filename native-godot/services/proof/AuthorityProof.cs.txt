string? Difference(JsonNode? a,JsonNode? b,string path="$"){
 if(a==null||b==null)return a==b?null:path+" null differs";
 if(a is JsonObject aa&&b is JsonObject bb){foreach(var k in aa.Select(p=>p.Key).Union(bb.Select(p=>p.Key))){var d=Difference(aa[k],bb[k],path+"."+k);if(d!=null)return d;}return null;}
 if(a is JsonArray ar&&b is JsonArray br){if(ar.Count!=br.Count)return path+" array lengths "+ar.Count+"/"+br.Count;for(int i=0;i<ar.Count;i++){var d=Difference(ar[i],br[i],path+"["+i+"]");if(d!=null)return d;}return null;}
 if(a.GetValueKind()==JsonValueKind.Number&&b.GetValueKind()==JsonValueKind.Number)return Math.Abs(N(a)-N(b))<1e-9?null:path+" "+a+" != "+b;
 return JsonNode.DeepEquals(a,b)?null:path+" "+a+" != "+b;
}
