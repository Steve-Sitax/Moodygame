 {Urls.Add(request.RequestUri!.ToString());if(request.Headers.Contains("x-api-key")||request.Headers.Contains("Authorization"))KeyedUrls.Add(request.RequestUri!.ToString());

  Body=JsonNode.Parse(await request.Content!.ReadAsStringAsync(stop));HasAuthorization=request.Headers.Contains("Authorization");return new HttpResponseMessage(HttpStatusCode.OK){Content=new StringContent(Reply,Encoding.UTF8,"application/json")};}
