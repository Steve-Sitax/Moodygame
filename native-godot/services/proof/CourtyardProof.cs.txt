using Scheldemist.People;using System.Text.Json.Nodes;using System.Globalization;
string root=Path.GetFullPath(args[0]);var data=JsonNode.Parse(File.ReadAllText(Path.Combine(root,"native-godot/services/proof/courtyard-oracles.json")))!;int cases=0;
double N(JsonNode? n)=>n==null?0:double.Parse(n.ToJsonString(),CultureInfo.InvariantCulture);
NativeCourtAnimal Animal(JsonNode n)=>new(){Id=n["id"]!.GetValue<string>(),Kind=n["kind"]!.GetValue<string>(),HomeX=N(n["hx"]),HomeZ=N(n["hz"]),X=N(n["x"]),Z=N(n["z"]),Yaw=N(n["yaw"]),Timer=N(n["t"]),TargetX=N(n["tx"]),TargetZ=N(n["tz"]),Peck=N(n["peck"]),Gait=N(n["gait"])};
void Assert(NativeCourtAnimal a,JsonNode n){foreach(var p in new[]{("hx",a.HomeX),("hz",a.HomeZ),("x",a.X),("z",a.Z),("yaw",a.Yaw),("t",a.Timer),("tx",a.TargetX),("tz",a.TargetZ),("peck",a.Peck),("gait",a.Gait)})if(Math.Abs(p.Item2-N(n[p.Item1]))>1e-9)throw new Exception("Case "+cases+" "+p.Item1+" "+p.Item2+" != "+n[p.Item1]);if(a.Id!=n["id"]!.GetValue<string>())throw new Exception("Id differs");}
foreach(var row in data["selections"]!.AsArray()){var homes=row!["homes"]!.AsArray().Select(n=>(Native:new NativeCourtHome((int)N(n!["house"]),N(n["x"]),N(n["z"]),N(n["ox"]),N(n["oz"])),Width:N(n["width"]),Blocked:n["blocked"]!.GetValue<bool>())).ToArray();int test=(int)N(row["test"]);
 var result=NativeCourtyardRules.Make(homes.Select(h=>h.Native),(x,z)=>{var h=homes.FirstOrDefault(h=>Math.Sqrt(Math.Pow(x-h.Native.X,2)+Math.Pow(z-h.Native.Z,2))<20);double dx=x-h.Native.X,dz=z-h.Native.Z,f=dx*h.Native.OutX+dz*h.Native.OutZ,s=-dx*h.Native.OutZ+dz*h.Native.OutX;return f>=0&&f<h.Width&&Math.Abs(s)<14;},(x,z,r)=>homes.Any(h=>Math.Sqrt(Math.Pow(x-h.Native.X,2)+Math.Pow(z-h.Native.Z,2))<20&&!h.Blocked),(x,z,r)=>!(test%3==0&&x>120));
 // The source stores hens/goats separately; selection still interleaves them for spacing.
 result=result.Where(a=>a.Kind=="hen").Concat(result.Where(a=>a.Kind=="goat")).ToList();var expected=row["result"]!.AsArray();if(result.Count!=expected.Count)throw new Exception("Selection count differs "+cases);for(int i=0;i<result.Count;i++)Assert(result[i],expected[i]!);cases++;
}
foreach(var row in data["motions"]!.AsArray()){int test=(int)N(row!["test"]);var a=Animal(row["initial"]!);uint seed=(uint)N(row["rngSeed"]);double Random(){seed=unchecked(seed*1664525+1013904223);return seed/4294967296.0;}bool Free(double x,double z,double r)=>test%3!=0||x*x+z*z>1;double dt=N(row["dt"]);if(a.Kind=="hen")NativeCourtyardRules.Hen(a,dt,row["players"]!.AsArray().Select(p=>(N(p!["x"]),N(p["z"]))).ToArray(),Random,Free);else NativeCourtyardRules.Goat(a,dt,Random,Free);Assert(a,row["result"]!);var pose=row["pose"]!;if(a.Walking!=(pose["motion"]!.GetValue<string>()=="walk"))throw new Exception("Motion differs "+cases);cases++;}
Console.WriteLine(System.Text.Json.JsonSerializer.Serialize(new{passed=true,sourceCases=cases,selections=48,movements=700}));
