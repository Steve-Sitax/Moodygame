using Scheldemist.Net.Mp;using System.Text.Json.Nodes;using System.Globalization;
string project=Path.GetFullPath(args[0]);var rows=JsonNode.Parse(File.ReadAllText(Path.Combine(project,"native-godot/services/proof/extras-oracles.json")))!["rows"]!.AsArray();int count=0,checks=0;
double N(JsonNode? n)=>n==null?0:double.Parse(n.ToJsonString(),CultureInfo.InvariantCulture);
NativeFigurePose Fig(JsonNode n)=>new((ushort)N(n["id"]),n["kind"]!.GetValue<string>(),n["motion"]!.GetValue<string>(),n["snap"]!.GetValue<bool>(),n["carrying"]!.GetValue<bool>(),N(n["x"]),N(n["y"]),N(n["z"]),N(n["yaw"]),N(n["speed"]));
void Check(bool pass,string message){if(!pass)throw new Exception(message);checks++;}
foreach(var n in rows){string type=n!["type"]!.GetValue<string>();if(type=="wire"){
 var fig=NativeExtraProtocol.EncodeFigures(N(n["time"]),new[]{Fig(n["fig"]!)},2);Check(Convert.ToHexString(fig).ToLowerInvariant()==n["figHex"]!.GetValue<string>(),"Figure wire "+count);var a=n["animal"]!;var an=NativeExtraProtocol.EncodeAnimals(N(n["time"]),new[]{((ushort)N(a["num"]),new NativeAnimalPose(N(a["x"]),N(a["z"]),N(a["yaw"]),a["motion"]!.GetValue<string>(),a["snap"]!.GetValue<bool>()),a["dog"]!.GetValue<bool>())});Check(Convert.ToHexString(an).ToLowerInvariant()==n["animalHex"]!.GetValue<string>(),"Animal wire "+count);
 Check(NativeExtraProtocol.ReadFigures(fig,out var sender,out var t,out var fs)&&sender==2&&t==N(n["time"])&&fs.Count==1,"Figure read");var expected=n["decodedFig"]!["list"]![0]!;Check(Math.Abs(fs[0].Yaw-N(expected["yaw"]))<1e-12&&Math.Abs(fs[0].Speed-N(expected["speed"]))<1e-12,"Figure decoding");
 Check(NativeExtraProtocol.ReadAnimals(an,out var at,out var animals)&&at==N(n["time"])&&animals.Count==1,"Animal read");var ea=n["decodedAnimal"]!["list"]![0]!;Check(Math.Abs(animals[0].Pose.Yaw-N(ea["yaw"]))<1e-12&&animals[0].Pose.X==N(ea["x"]),"Animal decoding");
 }else if(type=="delay")Check(NativeExtraProtocol.Delay(n["late"]!.AsArray().Select(v=>N(v)).ToArray())==N(n["expected"]),"Delay "+count);
 else{var frames=n["frames"]!.AsArray().Select(v=>(N(v!["t"]),Fig(v!))).ToArray();var got=NativeExtraProtocol.Sample(frames,N(n["time"]));var e=n["expected"]!;Check(got is{} s&&Math.Abs(s.X-N(e["x"]))<1e-9&&Math.Abs(s.Z-N(e["z"]))<1e-9&&Math.Abs(s.Yaw-N(e["yaw"]))<1e-9&&s.Snap==e["snap"]!.GetValue<bool>()&&s.Motion==e["motion"]!.GetValue<string>()&&s.Carrying==e["carrying"]!.GetValue<bool>(),"Interpolation "+count);}count++;
}
var stream=new NativeFigureReplication();var key=new object();var state=new NativeFigurePose(0,"thief","idle",false,false,1,0,2,0,0);stream.Register(key,()=>state);var sent=new List<byte[]>();bool Send(byte[] bytes){sent.Add(bytes);return true;}
stream.Send(.01,10000,Send);Check(sent.Count==1,"New figure sends immediately");stream.Send(.1,10100,Send);Check(sent.Count==1,"Standing cadence waits");stream.Send(.4,10500,Send);Check(sent.Count==2,"Standing sends twice per second");state=state with{Speed=1,Motion="walk"};stream.Send(.1,10600,Send);Check(sent.Count==3,"Moving sends ten per second");
stream.Remove(key);stream.Send(.01,10610,Send);stream.Send(.5,11110,Send);stream.Send(.5,11610,Send);stream.Send(.5,12110,Send);Check(sent.Count==6&&sent.Skip(3).All(b=>b[1]==0),"Exactly three empty removal batches");
var receiver=new NativeFigureReplication();receiver.Receive(NativeExtraProtocol.EncodeFigures(10000,new[]{state with{Id=3}},2),10010);Check(receiver.Frame(.1,10210).Count==1,"Remote figure appears");receiver.Receive(NativeExtraProtocol.EncodeFigures(10300,Array.Empty<NativeFigurePose>(),2),10310);Check(receiver.Frame(.1,10400).Count==1,"Missing figure waits for drawn batch");Check(receiver.Frame(.1,10501).Count==0,"Missing figure disappears at drawn time");
receiver.Receive(NativeExtraProtocol.EncodeFigures(11000,new[]{state with{Id=3}},2),11010);Check(receiver.Frame(.1,13600).Count==0,"Silent sender expires");receiver.Receive(NativeExtraProtocol.EncodeFigures(14000,new[]{state with{Id=3}},2),14010);receiver.DropSender(2);Check(receiver.Frame(.1,14010).Count==0,"Disconnect removes sender immediately");
receiver.Receive(NativeExtraProtocol.EncodeFigures(15000,new[]{state with{Id=3}},2),15010);receiver.Reset();Check(receiver.Frame(.1,15010).Count==0,"Reconnect clears old replicas");stream.Register(key,()=>state);stream.Reset();sent.Clear();stream.Send(0,16000,Send);Check(sent.Count==1&&NativeExtraProtocol.ReadFigures(sent[0],out _,out _,out var reset)&&reset[0].Snap,"Reconnect forces fresh snapshot");
Console.WriteLine(System.Text.Json.JsonSerializer.Serialize(new{passed=true,sourceCases=count,checks}));
