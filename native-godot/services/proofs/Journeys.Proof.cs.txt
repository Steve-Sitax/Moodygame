using System.Text.Json.Nodes;

double N(JsonNode? n)=>n?.GetValue<double>()??0;string S(JsonNode? n)=>n?.GetValue<string>()??"";bool B(JsonNode? n)=>n?.GetValue<bool>()??false;
JourneyRules.Point P(JsonNode n)=>new(N(n[0]),N(n[1]));
foreach(var row in oracle["modes"]!.AsArray()){

 var got=JourneyRules.Choose(new(P(a["from"]!),P(a["to"]!),load,(int)N(a["group"]),B(a["has"]!["velocipede"]),B(a["has"]!["handcart"]),B(a["has"]!["dray"]),B(a["has"]!["boat"]),a["water"]==null?null:P(a["water"]!["from"]!),a["water"]==null?null:P(a["water"]!["to"]!),stops,B(row["useWait"])?s=>N(row["waits"]![s.Id]):null,(int)N(a["age"])));
 var expected=row["result"]!;Check(got.Mode==S(expected["mode"])&&got.Why==S(expected["why"])&&Math.Abs(got.Dist-N(expected["dist"]))<1e-8&&Math.Abs(got.Secs-N(expected["secs"]))<1e-8&&got.FareC==N(expected["fare_c"])&&(got.Board?.Id??"")==S(expected["bus"]?["board"]?["id"])&&(got.Alight?.Id??"")==S(expected["bus"]?["alight"]?["id"]),"source mode choice (load/age/family/stop/arrival priority)");
}
foreach(var row in oracle["loading"]!.AsArray()){
 var got=JourneyRules.PlanLoading((int)N(row!["items"]),"owner",row["family"]!.AsArray().Select(f=>new JourneyRules.Family(S(f!["id"]),(int)N(f["age"]),B(f["home"]))));var expected=row["result"]!;

}

}
