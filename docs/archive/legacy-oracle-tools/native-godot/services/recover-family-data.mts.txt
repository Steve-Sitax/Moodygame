import {readFileSync,writeFileSync} from "node:fs";
import {resolve} from "node:path";
import {createHash} from "node:crypto";
import {runInNewContext} from "node:vm";
import {openDb} from "../../server/src/db.ts";
import {town} from "../../server/src/town/store.ts";
import {FORTUNE_ID,STRANGER_IDS} from "../../server/src/town/visitors.ts";
process.env.SCHELDEMIST_TOWN_SEED="1873";process.env.SCHELDEMIST_NO_AI="1";
const db=openDb(":memory:"),out=resolve(process.argv[2]??"native-godot/services/data");
const text=readFileSync("server/src/director/surprises.ts","utf8");
function literal(name:string){const start=text.indexOf("const "+name);if(start<0)throw new Error(name);const equals=text.indexOf("= {",start)+2;const end=text.indexOf("\n};",equals)+2;return runInNewContext("("+text.slice(equals,end)+")",Object.create(null),{timeout:1000});}
const prototypes=town(db).town.residents.filter(r=>[FORTUNE_ID,...STRANGER_IDS].includes(r.id));

writeFileSync(resolve(out,"native-family.json"),JSON.stringify(facts,null,2)+"\n");
