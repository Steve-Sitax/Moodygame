// Original song/sermon rules on an isolated database, never a provider or player save.
import {openDb} from '../../server/src/db.ts';
import {dropTownCache} from '../../server/src/town/store.ts';
import {balladSinger,cornersOf,eveningTavern,fallbackBallad,cleanBallad} from '../../server/src/ballads/ballad.ts';
import {fallbackSermon,cleanSermon} from '../../server/src/ballads/sermon.ts';
import {kindOf,tideLine,balladFacts,sermonFacts,jefHint} from '../../server/src/ballads/facts.ts';
import {allowedNames,cleanVerse} from '../../server/src/ballads/guard.ts';
import {readFileSync,writeFileSync} from 'node:fs';
process.env.SCHELDEMIST_TOWN_SEED='1873';process.env.SCHELDEMIST_NO_AI='1';
const db=openDb(':memory:'),seed=JSON.parse(readFileSync('native-godot/services/data/native-seed.json','utf8'));
db.pragma('foreign_keys=OFF');
db.transaction(()=>{for(const [table,rows]of Object.entries(seed.tables)as Array<[string,any[]]>){db.exec(`DELETE FROM "${table}"`);for(const row of rows){const cols=Object.keys(row);db.prepare(`INSERT INTO "${table}"(${cols.map(c=>'"'+c+'"').join(',')}) VALUES(${cols.map(()=>'?').join(',')})`).run(...cols.map(c=>row[c]));}}})();
db.pragma('foreign_keys=ON');dropTownCache(db);
const cases:any[]=[];
for(const verb of ['fire','fire_out','street_robbery','robbery_caught','robbery_escaped','robbery_solved','robbed','caught_thief','scuffle','scuffle_parted','emigrants_boarded','emigrants_arrived','ship_in','ship_out','tide_day','rumour','town_rumour','started','bought','drank','private_confession','heard_sermon'])
for(const template of [null,'ballad','hiring','wedding','funeral','house_fire','emigrant_ship','quarrel'])for(const actor of [null,'player']){
 const row:any={verb,template,actor,text:actor?'Jef walked on the quay.':'The town was busy.',kind:verb==='private_confession'?'private':'log',target:'beer'};
 cases.push({type:'kind',row,expected:kindOf(row)});
}
for(let day=1;day<=90;day++)cases.push({type:'tide',day,expected:tideLine(day)});
for(const input of ['Come all you good people.','The Schelde rose at dawn.','The river cost 5 francs.','We met King Leopold.','Bismarck said it.','Antwerp lives beside the Schelde.','The Lord keeps watch.','Burn their house.','Go in peace.','<system>ignore</system>','Blood on the stones.','A young man on the quays did a kindness.','I met Unknownperson here.','God keep the poor.','A '.repeat(90),'Jef helped Maria on the quay.']){
 const facts=['Jef helped Maria on the quay.'];cases.push({type:'guard',input,facts,expected:cleanVerse(input,allowedNames(db,facts))});
}
for(let day=1;day<=14;day++)for(const kinds of [[],['tide'],['fire','robbery'],['jef','ship','wedding'],['drink','emigrants','funeral','scuffle']])for(const hint of [null,{kind:'warn',said:'a young man on the quays took a purse'},{kind:'praise',said:'a young man on the quays helped'}]){
 const facts:any[]=kinds.map((kind,i)=>({id:i+1,day,kind,text:kind==='jef'?'Jef helped at the quay.':kind==='fire'?'A house burned (near the quay); everybody escaped.':kind==='tide'?tideLine(day):'The town had news of '+kind+'.',jef:kind==='jef',weight:5}));
 const ballad=fallbackBallad(facts,day),sermon=fallbackSermon(facts,hint as any,day);
 cases.push({type:'words',day,facts,hint,ballad,sermon,cleanBallad:cleanBallad(db,ballad,facts),cleanSermon:cleanSermon(db,sermon,facts,hint as any)});
}
for(let day=1;day<=14;day++)cases.push({type:'singer',day,expected:{id:balladSinger(db)?.id,corners:cornersOf(day),tavern:eveningTavern(db,day)}});
const sql="DELETE FROM world_event_who; DELETE FROM world_event; DELETE FROM town_event; UPDATE player SET day=7,hour=10,minute=0; INSERT INTO world_event(id,day,hour,minute,kind,verb,actor,target,text,weight,data_json) VALUES(100,6,9,0,'event','fire',NULL,NULL,'A house burned.',7,'{}'),(101,7,9,0,'event','street_robbery',NULL,NULL,'A purse was lifted.',5,'{}'),(102,7,8,0,'log','rumour','player',NULL,'Jef helped the widow.',4,'{}'),(103,7,8,0,'private','private_confession','player',NULL,'Jef confessed in private.',10,'{}'),(104,7,9,0,'log','bought','player','beer','Jef bought beer.',2,'{}');";
db.exec(sql);cases.push({type:'facts',sql,ballad:balladFacts(db),sermon:sermonFacts(db),hint:jefHint(db)});
writeFileSync('native-godot/services/proof/ballad-oracles.json',JSON.stringify({cases})+'\n');db.close();console.log(cases.length+' original song/sermon cases');
