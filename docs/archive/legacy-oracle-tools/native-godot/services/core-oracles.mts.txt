import { openDb } from "../../server/src/db.ts";
import { settle } from "../../server/src/game.ts";

const db=openDb(":memory:");
WHERE.now=()=>({shelter:"outside",lantern:false,place:null,label:"outside"});

  applyHour(db,hour);cases.push({type:"hour",initial,hour,expected:needs(db)});
}
const tasks=[
  {kind:"carry",goods:"crates",count:2,from:"pier_head",to:"crane_foot",twist:"none",limit_s:null},
  {kind:"carry",goods:"sacks",count:2,from:"pier_head",to:"crane_foot",twist:"foreman_watches",limit_s:null},

  {kind:"watch",goods:"barrels",post:"west_sheds",duration_s:90,twist:"foreman_watches"},
  {kind:"letters",stops:[{what:"letter",done:true},{what:"letter",done:false}]},

  {kind:"lamps",lamps:[{done:false}]},
  {kind:"mill",turns:3},
];
for(const task of tasks)for(const roll of [0,.349,.35,.99])for(let variant=0;variant<18;variant++){
 const total=task.kind==="carry"?task.count:1;
 const report:any={delivered:variant%3===0?0:total,lost:variant%3===0?total:0,sold:0,pocketed:variant%4===0,late:variant%5===0,left_post_s:variant%3===1?21:0,thief:variant%3===0?"none":variant%3===1?"chased":"stole",bribe_taken:variant%2===0,seen_away:variant%7===0,turns:variant%4};

 const job:any={id:1,title:"Oracle",employer_npc:"sooi",employer_name:"Sooi",pay_c:95,task_type:task.kind,task};
 try{cases.push({type:"settle",job,report,roll,expected:settle(job,report,()=>roll)});}catch(error:any){cases.push({type:"settle",job,report,roll,error:error.message,status:error.status});}
}
for(const text of ["Jef stole a herring from Rosalie's stall, and Rosalie saw it.","Rosalie Peeters says you took it.","He gave it back to Rosalie.","Rosalie is angry.","Jef took Fientje's herring.","Rosalind was there."]){const names=["Rosalie Peeters","Rosalie"];cases.push({type:"ownVoice",text,names,expected:ownVoice(text,names)});}
for(const fact of ["Jef rents a room from Rosalie.","Jef carries two sacks for Sooi.","Jef goes to the town hall.","Jef was rude to Anna.","Jef has his mother's medal.","Jef did honest work for Sooi.","Jef's coat is wet.","Somebody asked after work."]){cases.push({type:"toYou",fact,expected:toYou(fact)});cases.push({type:"whoYou",fact,expected:whoYou(fact)});}
for(const fact of ["Jef stole two herrings from Fientje.","Jef helped Rosalie.","Jef did honest work for Sooi.","Jef was rude to Anna.","Jef carried two sacks for Sooi.","Jef gave a herring to Fientje.","Jef asked after work."]){
 for(const roll of [0,.1,.349,.35,.6,.99])cases.push({type:"drift",fact,told:fact,roll,expected:driftWording(fact,fact,()=>roll)});
 for(const told of [fact,"Jef murdered Anna.","Jef stole fifty sacks from Sooi.","Jef never did honest work for Sooi.","Jef gave a herring to Fientje, they say.","Jef did honest work for Sooi, or so they say"] )cases.push({type:"withinFact",fact,told,expected:withinFact(fact,told)});
}
for(const npc of ["sooi","fientje","peeters","r001"] )for(const roll of [0,.349,.35,.99]){

 const heard=spreadRumours(db,()=>roll);const expected=db.prepare("SELECT npc_id,text,source,heard_from,weight,gist,tone,origin,town_spread,told_as,about_player FROM npc_memory ORDER BY id").all();cases.push({type:"spread",initial,roll,heard,expected});
}

 cases.push({type:"homeFirstFit",cls,placed:[],kind,expected:firstFit(cls,[],kind)});
}

  cases.push({type:"homeComfort",cls,placed,expected:comfortOf(cls,placed)});cases.push({type:"homeNight",cls,placed,expected:nightAt(cls,placed)});

 }
