import {writeFileSync,readFileSync} from "node:fs";
import {createHash} from "node:crypto";
import {openDb} from "../../server/src/db.ts";
import {town} from "../../server/src/town/store.ts";
import {POSTS,WARE_GOOD,LAST_LINE,RUN_MAX,postOpen,stockPrice,demandFactor,runAt} from "../../shared/trade.ts";
import {MILLS} from "../../shared/mills.ts";
import {HAUL_ROUTES} from "../../shared/hauls.ts";
import {drovePigs} from "../../shared/drove.ts";
import {stepLedger,settleRuns,tradeLedger,writeLedger,postDoor} from "../../server/src/trade/ledger.ts";
import {installTrade} from "../../server/src/trade/routes.ts";
import {waresOf} from "../../server/src/trade.ts";
process.env.SCHELDEMIST_TOWN_SEED="1873";process.env.SCHELDEMIST_NO_AI="1";
const facts={version:1,posts:POSTS,wareGood:WARE_GOOD,lastLine:LAST_LINE,runMax:RUN_MAX,mills:MILLS,hauls:HAUL_ROUTES,spots:JSON.parse(readFileSync("shared/spots.json","utf8"))};
writeFileSync("native-godot/services/data/native-trade-ledger.json",JSON.stringify(facts,null,2)+"\n");
const prices=[];for(const post of POSTS)for(const list of [1,5,9,10,15,37,100])for(const stock of [0,post.floor,post.floor+.1,post.order-1,post.order,post.room*.85,post.room])for(const hour of [6,12,13,15,18])prices.push({post:post.id,list,stock,hour,expected:stockPrice(list,post,stock,hour)});
const openings=[];for(const post of POSTS)for(const day of [1,6,7,14])for(const hour of [0,1,2,5,5.5,6,7,9.5,12,17,18,18.5,23])openings.push({post:post.id,day,hour,expected:postOpen(post,day,hour)});
const steps=[];for(const start of [0,345,360,410,1435,6*1440+350,7*1440+350])for(const gap of [4,5,120,1440,3*1440+120]){const ledger={at:start,stock:Object.fromEntries(POSTS.map(p=>[p.id,p.start])),sold:{}};stepLedger(ledger,start+gap);steps.push({start,gap,expected:ledger});}
const runs=[];for(const len of [0,3,120,360])for(const elapsed of [-1,0,5.9,6,7,10,20,60,120,400]){const run={id:"run",good:"bread",from:"bakery_rijn",to:"bakery_steen",n:12,man:"r001",t0:360,way:[[0,0],[10,10]],len,done:false};runs.push({run,elapsed,expected:runAt(run as any,360+elapsed)});}
const pigs=Array.from({length:60},(_,i)=>({day:i+1,expected:drovePigs(i+1)}));
const db=openDb(":memory:");installTrade();const seller=town(db).town.shops.find(s=>s.id==="bakery_steen")!.keeper;
const market=[];for(const npc of ["fientje",seller])for(const factor of [.5,1,1.5,2,3])for(const stock of [0,4,14,80]){const kind=npc==="fientje"?"herring":"bread",post=npc==="fientje"?"vismarkt":"bakery_steen";db.prepare("INSERT INTO world_state(key,value_json)VALUES('m4_prices',?)ON CONFLICT(key)DO UPDATE SET value_json=excluded.value_json").run(JSON.stringify({[kind]:{factor}}));const l=tradeLedger(db);l.stock[post]=stock;writeLedger(db,l);market.push({npc,kind,post,factor,stock,expected:waresOf(db,npc).find(w=>w.kind===kind)!.price_c});}
writeFileSync("native-godot/services/proofs/data/native-trade-ledger-source.json",JSON.stringify({prices,openings,steps,runs,pigs,market},null,2)+"\n");
const paths=["server/src/trade/routes.ts","server/src/trade/ledger.ts","server/src/trade/rush.ts","server/src/trade/drove.ts","shared/trade.ts","shared/drove.ts"];writeFileSync("native-godot/services/data/native-trade-ledger-intake.json",JSON.stringify({version:1,hashes:Object.fromEntries(paths.map(p=>[p,createHash("sha256").update(readFileSync(p)).digest("hex")]))},null,2)+"\n");db.close();console.log(JSON.stringify({prices:prices.length,openings:openings.length,steps:steps.length,runs:runs.length,pigs:pigs.length,market:market.length}));
