import {openDb} from '../../server/src/db.ts';
import {fallbackBoard,boardSize,gateCarts,clampBoard,FALLBACK_BOARD} from '../../server/src/hooks/jobBoard.ts';
import {workLine,fallbackOutcome} from '../../server/src/hooks/jobOutcome.ts';

process.env.SCHELDEMIST_TOWN_SEED='1873';process.env.SCHELDEMIST_NO_AI='1';const db=openDb(':memory:');const cases:any[]=[];

const tasks=[{kind:'carry',goods:'sacks',count:2,from:'pier_head',to:'crane_foot',cart:false},{kind:'carry',goods:'crates',count:5,from:'crane_foot',to:'hessenatie_door',cart:true},{kind:'deliver',goods:'parcel',from:'peeters_dock',to:'ship_gangway',recipient:'the cook'},{kind:'deliver',goods:'rope',from:'pier_head',to:'east_carts',recipient:'the carter'},{kind:'watch',goods:'barrels',post:'west_sheds'},{kind:'mill',post:{label:'the mill'}},{kind:'lamps',lamps:[{},{}]},{kind:'letters',stops:[{what:'telegraph'}]},{kind:'letters',stops:[{what:'letter'},{what:'letter'}]},null];for(const task of tasks){const job:any={employer_name:'Sooi',task};cases.push({type:'workLine',job,expected:workLine(job)});for(const pay_c of [0,5,90])for(const caught of [false,true]){const settlement:any={pay_c,caught,facts:['Jef did the work.']};cases.push({type:'fallbackOutcome',job,settlement,expected:fallbackOutcome(job,settlement)});}}
